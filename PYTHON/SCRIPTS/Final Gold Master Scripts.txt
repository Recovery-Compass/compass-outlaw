### **PART 1: THE MASTER TEMPLATES (PYTHON)**

#### **TEMPLATE A: CALIFORNIA PLEADING PAPER (The "One Legal" Standard)**
*Use this for Motions, Petitions, Declarations, and Orders submitted to the Court.*
*   **Key Features:** 28-Line Grid, Double Left Rail, Single Right Rail, Caption Box, Centered Footer, Hard-coded Page Breaks.

```python
import io, textwrap
from reportlab.pdfgen import canvas
from reportlab.lib.pagesizes import LETTER

# --- PHYSICS ENGINE: CALIFORNIA PLEADING STANDARD (v15.0) ---
LEADING = 24.0
LINE_1_Y = 720.0
LEFT_RAIL_X = 85.0
TEXT_START_X = 95.0
CENTER_X = 306.0
RIGHT_RAIL_X = 560.0
CAPTION_LINE_X = 290.0

def get_line_y(line_num):
    # -4pt drop aligns text baseline to the numbered grid line
    return (LINE_1_Y - ((line_num - 1) * LEADING)) - 4

def draw_architecture(c, page_num):
    # Left Rails (Double)
    c.setLineWidth(1.0)
    c.line(LEFT_RAIL_X, 50, LEFT_RAIL_X, 750)
    c.line(LEFT_RAIL_X - 6, 50, LEFT_RAIL_X - 6, 750)
    
    # Right Rail (Single) - Starts at Caption top on P1, Page top on P2
    start_y = get_line_y(8) if page_num == 1 else 750
    c.line(RIGHT_RAIL_X, 50, RIGHT_RAIL_X, start_y)
    
    # Line Numbers (1-28)
    c.setFont("Courier", 10)
    for i in range(1, 29):
        c.drawRightString(LEFT_RAIL_X - 10, (LINE_1_Y - ((i - 1) * LEADING)) - 3, str(i))

    # Centered Footer
    c.setFont("Times-Roman", 9)
    footer_text = f"Case No: 25PR000590 | PETITION TITLE HERE | Page {page_num}"
    c.drawCentredString(CENTER_X, 30, footer_text)

def generate_pleading():
    filename = "CALIFORNIA_PLEADING_TEMPLATE.pdf"
    c = canvas.Canvas(filename, pagesize=LETTER)

    # ================= PAGE 1 =================
    draw_architecture(c, 1)

    # 1. HEADER (Attorney Info)
    c.setFont("Times-Roman", 12)
    header = [
        "ERIC BRAKEBILL JONES",
        "5634 Noel Drive",
        "Temple City, CA 91780",
        "(626) 348-3019",
        "Petitioner In Pro Per"
    ]
    start_y = LINE_1_Y - 5 
    for idx, line in enumerate(header):
        c.drawString(TEXT_START_X, start_y - (idx * LEADING), line)

    # 2. COURT TITLE
    c.setFont("Times-Bold", 12)
    c.drawCentredString(CENTER_X, get_line_y(8)+4, "SUPERIOR COURT OF THE STATE OF CALIFORNIA")
    c.drawCentredString(CENTER_X, get_line_y(9)+4, "FOR THE COUNTY OF MONTEREY")
    c.setFont("Times-Roman", 10)
    c.drawCentredString(CENTER_X, get_line_y(10)+8, "1200 Aguajito Road, Monterey, CA 93940")

    # 3. CAPTION BOX
    c.line(CAPTION_LINE_X, get_line_y(11) + 14, CAPTION_LINE_X, get_line_y(17) - 8)
    
    # Left Side (Parties)
    c.setFont("Times-Roman", 12)
    c.drawString(TEXT_START_X, get_line_y(11), "Estate of JUDY BRAKEBILL JONES,")
    c.drawString(TEXT_START_X, get_line_y(12), "Decedent.")
    
    # Right Side (Case Info)
    c.drawString(CAPTION_LINE_X + 10, get_line_y(11), "Case No.: 25PR000590")
    c.setFont("Times-Bold", 12)
    c.drawString(CAPTION_LINE_X + 10, get_line_y(12), "DOCUMENT TITLE")
    c.drawString(CAPTION_LINE_X + 10, get_line_y(13), "GOES HERE")
    c.drawString(CAPTION_LINE_X + 10, get_line_y(14), "(CODE SECTION)")
    
    c.setFont("Times-Roman", 12)
    c.drawString(CAPTION_LINE_X + 10, get_line_y(15), "Dept: 13")
    c.drawString(CAPTION_LINE_X + 10, get_line_y(16), "Judge: Hon. Elisabeth K. Mineta")
    c.drawString(CAPTION_LINE_X + 10, get_line_y(17), "Hearing: March 4, 2026")

    # Box Bottom Line
    c.line(TEXT_START_X, get_line_y(17) - 8, RIGHT_RAIL_X, get_line_y(17) - 8)

    # 4. PAGE 1 CONTENT (Manual Block Allocation)
    c.setFont("Times-Roman", 12)
    c.drawString(TEXT_START_X, get_line_y(19), "1. PETITIONER: I, ERIC BRAKEBILL JONES, am a party to this action")
    c.drawString(TEXT_START_X + 15, get_line_y(20), "acting in Pro Per.")

    # 5. CONTINUED MARKER
    c.drawString(TEXT_START_X, get_line_y(27), "// (Continued on Page 2)")
    c.showPage()

    # ================= PAGE 2 =================
    draw_architecture(c, 2)
    c.setFont("Times-Bold", 12)
    c.drawCentredString(CENTER_X, get_line_y(1), "DOCUMENT TITLE (CONTINUED)")
    
    c.setFont("Times-Roman", 12)
    
    # Content Block
    p_text = "2. ARGUMENT: Insert legal argument here. Text wraps automatically to 70 characters."
    lines = textwrap.wrap(p_text, width=70)
    current_line = 3
    for line in lines:
        c.drawString(TEXT_START_X, get_line_y(current_line), line)
        current_line += 1

    # Signature Block
    current_line += 2
    c.drawString(TEXT_START_X, get_line_y(current_line), "DATED: January 12, 2026")
    c.line(300, get_line_y(current_line), 500, get_line_y(current_line))
    c.drawString(300, get_line_y(current_line+1), "ERIC BRAKEBILL JONES")
    c.drawString(300, get_line_y(current_line+2), "Petitioner In Pro Per")

    c.save()
    print(f"[+] TEMPLATE GENERATED: {filename}")

if __name__ == "__main__":
    generate_pleading()
```

#### **TEMPLATE B: RECORDING INSTRUMENT (The "Recorder" Standard)**
*Use this for Lis Pendens, Mechanic's Liens, or Affidavits to be recorded against property.*
*   **Key Features:** 2.5" Top Right Reserve, Notary Block, Anti-Collision Spacing.

```python
import io, textwrap
from reportlab.pdfgen import canvas
from reportlab.lib.pagesizes import LETTER

# --- PHYSICS ENGINE: RECORDER STANDARD v4.0 (ANTI-COLLISION) ---
LEADING = 14.0 
LEFT_MARGIN = 50
RIGHT_MARGIN = 560

def create_recording_doc():
    filename = "CALIFORNIA_RECORDING_TEMPLATE.pdf"
    c = canvas.Canvas(filename, pagesize=LETTER)

    # --- PAGE 1: THE NOTICE ---
    
    # 1. RECORDING REQUEST BLOCK (Top Left)
    c.setFont("Times-Roman", 10)
    c.drawString(LEFT_MARGIN, 750, "RECORDING REQUESTED BY:")
    c.drawString(LEFT_MARGIN, 740, "Eric Brakebill Jones")
    c.drawString(LEFT_MARGIN, 730, "5634 Noel Drive")
    c.drawString(LEFT_MARGIN, 720, "Temple City, CA 91780")
    
    c.drawString(LEFT_MARGIN, 700, "WHEN RECORDED MAIL TO:")
    c.drawString(LEFT_MARGIN, 690, "Eric Brakebill Jones")
    c.drawString(LEFT_MARGIN, 680, "5634 Noel Drive")
    c.drawString(LEFT_MARGIN, 670, "Temple City, CA 91780")
    
    # 2. TITLE (Centered - Below Recorder Space)
    c.setFont("Times-Bold", 14)
    c.drawCentredString(306, 580, "NOTICE OF PENDENCY OF ACTION (LIS PENDENS)")
    c.setFont("Times-Bold", 10)
    c.drawCentredString(306, 565, "(Probate Code § 1004; CCP § 405.20)")

    # 3. CAPTION
    y = 540
    c.setFont("Times-Roman", 12)
    c.drawString(LEFT_MARGIN, y, "SUPERIOR COURT OF CALIFORNIA, COUNTY OF MONTEREY")
    y -= 20
    c.drawString(LEFT_MARGIN, y, "Estate of JUDY BRAKEBILL JONES, Decedent.")
    c.drawString(350, y, "Case No.: 25PR000590")
    
    y -= 40
    
    # 4. BODY TEXT
    body_top = """NOTICE IS HEREBY GIVEN that an action has been commenced..."""
    text = c.beginText(LEFT_MARGIN, y)
    text.setFont("Times-Roman", 12)
    text.setLeading(16) 
    for para in body_top.split('\n'):
        lines = textwrap.wrap(para.strip(), width=80)
        for line in lines: text.textLine(line)
        text.textLine(" ")
    c.drawText(text)
    
    # 5. PROPERTY DESCRIPTION (Fixed Y to prevent collision)
    y = 260 
    c.setFont("Times-Bold", 12)
    c.drawString(LEFT_MARGIN, y, "ADDRESS: 17742 Berta Canyon Road, Salinas, CA 93907")
    c.drawString(LEFT_MARGIN, y-15, "APN: 125-133-039")
    
    # 6. SIGNATURE (Fixed Bottom Y)
    y_sig = 100
    c.setFont("Times-Roman", 12)
    c.drawString(LEFT_MARGIN, y_sig + 40, "DATED: January 12, 2026")
    c.line(LEFT_MARGIN, y_sig, LEFT_MARGIN + 250, y_sig)
    c.drawString(LEFT_MARGIN, y_sig-15, "ERIC BRAKEBILL JONES")
    c.drawString(LEFT_MARGIN, y_sig-30, "Petitioner / Trustee In Pro Per")
    c.showPage()

    # --- PAGE 2: NOTARY ACKNOWLEDGMENT ---
    c.setFont("Times-Bold", 14)
    c.drawCentredString(306, 750, "CALIFORNIA ALL-PURPOSE ACKNOWLEDGMENT")
    
    # Notary Disclaimer
    c.setFont("Times-Roman", 12)
    c.drawString(LEFT_MARGIN, 720, "A notary public or other officer completing this certificate verifies only the")
    c.drawString(LEFT_MARGIN, 705, "identity of the individual who signed the document to which this certificate")
    c.drawString(LEFT_MARGIN, 690, "is attached, and not the truthfulness, accuracy, or validity of that document.")
    
    # Location Box
    c.rect(LEFT_MARGIN - 5, 600, 250, 60) 
    c.drawString(LEFT_MARGIN, 640, "State of California")
    c.drawString(LEFT_MARGIN, 620, "County of Los Angeles")
    
    # Acknowledgment Text
    text = c.beginText(LEFT_MARGIN, 570)
    text.setFont("Times-Roman", 12)
    text.setLeading(16)
    ack_body = "On [DATE] before me, [NOTARY], personally appeared ERIC BRAKEBILL JONES..."
    for line in textwrap.wrap(ack_body, width=80): text.textLine(line)
    c.drawText(text)
    
    # Seal Area
    c.rect(350, 200, 150, 100)
    c.setFont("Times-Italic", 10)
    c.drawString(360, 250, "(Place Notary Seal Above)")

    c.save()
    print(f"[+] TEMPLATE GENERATED: {filename}")

if __name__ == "__main__":
    create_recording_doc()
```
